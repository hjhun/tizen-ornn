cmake_minimum_required(VERSION 3.16)
project(tizen_cross_example LANGUAGES C CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0)
add_executable(probe-c probe.c)
target_link_libraries(probe-c PRIVATE PkgConfig::GLIB)
add_executable(probe-cxx probe.cc)
target_compile_features(probe-cxx PRIVATE cxx_std_17)
enable_testing()
add_test(NAME c-glib COMMAND probe-c)
add_test(NAME cxx COMMAND probe-cxx)
